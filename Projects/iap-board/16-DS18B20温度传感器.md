# 代码展示

## main.c

~~~C
#include "STC15F2K60S2.h"
#include "intrins.h"
#include "onewire.h"
#include "absacc.h"
  
unsigned char SMGNoDot[]={0xc0, 0xf9, 0xa4, 0xb0, 0x99, 0x92, 0x82, 0xf8, 0x80, 0x90};
unsigned char SMGDot[]={0x40, 0x79, 0x24, 0x30, 0x19, 0x12, 0x02, 0x78, 0x00, 0x10};
unsigned int temp = 0;
  
void Delay1ms()     //@11.0592MHz
{
    unsigned char i, j;
  
    _nop_();
    _nop_();
    i = 11;

    j = 190;
    do
    {
        while (--j);
    } while (--i);
}
  
void DisplaySMG_Bit(unsigned char pos, unsigned char dat){
    XBYTE[0xE000] = 0xff;//消隐
    XBYTE[0xC000] = 0x01 << pos;//决定数码管显示位置
    XBYTE[0xE000] = dat;//决定数码管显示的相应的数字或字符
}


  
void Display_All(unsigned char dat){
    XBYTE[0xE000] = 0xff;//消隐，关闭所有数码管
    XBYTE[0xC000] = 0xff;//选择所有数码管
    XBYTE[0xE000] = dat;//所有数码管显示内容为dat（下面看其实也是清空这些内容）
}
  
void DisplaySMG_Temp(){
    DisplaySMG_Bit(7, SMGNoDot[temp % 10]);
    Delay1ms();
    DisplaySMG_Bit(6, SMGDot[(temp % 100)/10]);
    Delay1ms();
    DisplaySMG_Bit(5, SMGNoDot[temp / 100]);
    Delay1ms();
    DisplaySMG_Bit(4, 0xff);
    Delay1ms();
    DisplaySMG_Bit(3, 0xff);
    Delay1ms();
    DisplaySMG_Bit(2, 0xff);
    Delay1ms();
    DisplaySMG_Bit(1, 0xff);
    Delay1ms();
    DisplaySMG_Bit(0, 0xff);
    Delay1ms();//这里往上，只是为了显示一次数码管而已
  
    Display_All(0xff);/*结合上面的子函数可以开出，用到这里只是为了消除所有显示的内容，以免残留显示。*/
}
  
void Read_DS18B20_temp(){
    unsigned char LSB, MSB;
  
    init_ds18b20();//发送复位脉冲，检测传感器的存在。
    Write_DS18B20(0xcc);//发送跳过POM指令（适用于单设备总线）
    Write_DS18B20(0x44);//启用温度转换设备
    Delay1ms();//等待转换完成（实际上需要750ms，这里需要改进）
  
    init_ds18b20();//再次复位
    Write_DS18B20(0xcc);//再次跳过POM
    Write_DS18B20(0xbe);//发送读取暂存器命令
  
    LSB = Read_DS18B20();//读取温度低字节（bit0-3为小数部分）
    MSB = Read_DS18B20();//读取温度高字节（bit4-10为整数部分）
  
    temp = MSB;
    temp = (MSB << 8) | LSB;//合并为16位温度值（补码）
  
    if ((temp & 0xf800) == 0x0000){//判断符号位（0为零上）
        temp = temp >> 4;//右移4位获取整数部分
        temp = temp * 10;//转化为十进制整数位，如25→250
        temp = temp + (LSB & 0x0f) * 0.625;//计算小数位
    }
}

void main(){
    XBYTE[8000] = 0xff;
    while(1){
        Read_DS18B20_temp();
        DisplaySMG_Temp();
    }
}
~~~

#include "STC15F2K60S2.H" /*这个头文件是需要自己加上去的*/
  
sbit DQ = P1^4;//这个引脚也是我自己定义的

//用于生成总线通信所需的延时，t是延迟的参数，通过循环来计时。没什么好说的
void Delay_OneWire(unsigned int t)  
{
    unsigned char i;
    while(t--){
        for(i=0;i<12;i++);
    }
}
  
//这个函数用于向DS18B20写入一个字节的数据。通过循环8次，每次写入一个位，最后再延时一段时间。
void Write_DS18B20(unsigned char dat)
{
    unsigned char i;
    for(i=0;i<8;i++)
    {
        DQ = 0;
        DQ = dat&0x01;
        Delay_OneWire(5);
        DQ = 1;
        dat >>= 1;
    }
    Delay_OneWire(5);
}
  
//这个函数用于从DS18B20读取一个字节的数据。通过循环8次，每次读取一个位，最后返回读取的数据。
unsigned char Read_DS18B20(void)
{
    unsigned char i;
    unsigned char dat;
    for(i=0;i<8;i++)
    {
        DQ = 0;
        dat >>= 1;
        DQ = 1;
        if(DQ)
        {
            dat |= 0x80;
        }      
        Delay_OneWire(5);
    }
    return dat;
}
  
/*这个函数用于初始化DS18B20传感器。通过拉低和拉高数据线来发送复位脉冲，并检测传感器的响应。如果传感器响应，initflag 为0，否则为1。*/
bit init_ds18b20(void)
{
    bit initflag = 0;//这里默认是0。
    DQ = 1;
    Delay_OneWire(12);
    DQ = 0;
    Delay_OneWire(80);
    DQ = 1;
    Delay_OneWire(10);//往上是拉低和拉高数据来产生复位脉冲。
    initflag = DQ;    //这里进行赋值，来检测传感器是否响应。
    Delay_OneWire(5);
    return initflag;
}
