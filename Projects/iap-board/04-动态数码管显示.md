# 代码展示

~~~C
#include "STC15F2K60S2.h" //这里使用了它专用的头文件
#include <intrins.h>  

unsigned char code SMG_duanma[18] = {
    0xc0,0xf9,0xa4,0xb0,0x99,0x92,
    0x82,0xf8,0x80,0x90,0x88,0x80,
    0xc6,0xc0,0x86,0x8e,0xbf,0x7f
};
  
void SelectHC573(unsigned char channel){
    switch(channel){
        case 0:
            P2 = (P2 & 0x1f) | 0x80;
            break;
        case 1:
            P2 = (P2 & 0x1f) | 0xa0;
            break;
        case 2:
            P2 = (P2 & 0x1f) | 0xc0;
            break;
        case 3:
            P2 = (P2 & 0x1f) | 0xe0;
            break;
    }
}
  
void Delay1ms(void) //@11.0592MHz
{
    unsigned char data i, j;
  
    _nop_();
    _nop_();
    _nop_();
    i = 11;
    j = 190;
    do
    {
        while (--j);
    } while (--i);
}//延迟1ms的模块，消影的同时显示多个内容。
  
void DisplaySMG_Bit(unsigned char value,unsigned char pos){
    SelectHC573(2);//数码管位选模式
    P0 = 0x01 << pos;
    SelectHC573(3);//数码管段选模式
    P0 = value;
}//这里显示数码管的实际功能
  
void Display_Dynamic(){
    DisplaySMG_Bit(SMG_duanma[1],0);
    Delay1ms();
    DisplaySMG_Bit(SMG_duanma[9],1);
    Delay1ms();
    DisplaySMG_Bit(SMG_duanma[1],2);
    Delay1ms();
    DisplaySMG_Bit(SMG_duanma[9],3);
    Delay1ms();
    DisplaySMG_Bit(SMG_duanma[0],4);
    Delay1ms();
    DisplaySMG_Bit(SMG_duanma[8],5);
    Delay1ms();
    DisplaySMG_Bit(SMG_duanma[1],6);
    Delay1ms();
    DisplaySMG_Bit(SMG_duanma[0],7);
    Delay1ms();
}//这里就自由发挥了，我这里显示了19190810
  
void main(){
    while(1){
        Display_Dynamic();
    }
}
~~~
