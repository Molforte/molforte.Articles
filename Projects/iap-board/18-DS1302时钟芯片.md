基本操作：其一是设定时间参数，其二是读取实时时间。

# 代码

## main.c

~~~C
#include "STC15F2K60S2.h"
#include "intrins.h"
#include "ds1302.h"
#include "smg_ca.h"//七段数码管显示的头文件。

/*DS1302的写地址数组，用于写入秒、分、时、日、月、星期、年。*/
unsigned char Write_DS1302_addr[7] = {0x80,0x82,0x84,0x86,0x88,0x8a,0x8c};

/*DS1302的读地址数组，用于读取秒、分、时、日、月、星期、年。*/
unsigned char Read_DS1302_addr[7] = {0x81,0x83,0x85,0x87,0x89,0x8b,0x8d};
/*20年4月18日，周六，23点59分24秒。
存储当前时间的数组，格式为秒、分、时、日、月、星期、年。*/
unsigned char Timer[7] = {0x24,0x59,0x23,0x18,0x04,0x06,0x20};
  
void DS1302_Config(){//DS1302配置函数。
    char i;
    Write_Ds1302_Byte(0x8e,0x00);//允许写入DS1302。
    for (i = 0; i < 7; i++)
    {
        Write_Ds1302_Byte(Write_DS1302_addr[i],Timer[i]);//写入数据到DS1302
    }
    Write_Ds1302_Byte(0x8e,0x80);//禁止写入DS1302。
}

/*从DS1302读取时间数据并存储到Timer数组中。
这个还是看得懂的hhh*/
void Read_DS1302_Timer(){
    char i;
    for (i = 0; i < 7; i++)
    {
        Timer[i] = Read_Ds1302_Byte(Read_DS1302_addr[i]);
    }
}
  
void Display_DS1302(){//用来显示时、分、秒
    DisplaySMG(0,SMGNoDot[Timer[2] /16]);
    Delay1ms();
    DisplaySMG(1,SMGNoDot[Timer[2] % 16]);
    Delay1ms();
  
    DisplaySMG(2,0xbf);
    Delay1ms();
  
    DisplaySMG(3,SMGNoDot[Timer[1] /16]);
    Delay1ms();
    DisplaySMG(4,SMGNoDot[Timer[1] % 16]);
    Delay1ms();
  
    DisplaySMG(5,0xbf);
    Delay1ms();
  
    DisplaySMG(6,SMGNoDot[Timer[0] /16]);
    Delay1ms();
    DisplaySMG(7,SMGNoDot[Timer[0] % 16]);
    Delay1ms();
  
    Display_All(0xff);
  
}
  
void main(){
    DS1302_Config();
    while(1){
        Read_DS1302_Timer();
        Display_DS1302();
    }
}
~~~

## ds1302.c

~~~C
/*  #   DS1302代码片段说明
    1.  本文件夹中提供的驱动代码供参赛选手完成程序设计参考。
    2.  参赛选手可以自行编写相关代码或以该代码为基础，根据所选单片机类型、运行速度和试题
        中对单片机时钟频率的要求，进行代码调试和修改。
*/                              
  
//
#include "STC15F2K60S2.h"
#include "intrins.h"
  
sbit SCK=P1^7;
sbit SDA=P2^3;
sbit RST=P1^3;
  
//以上的定义和声明都是我自己加的，蓝桥杯比赛可能不会提供这些定义和声明，需要自己定义和声明。
  
void Write_Ds1302(unsigned  char temp)
{
    unsigned char i;
    for (i=0;i<8;i++)      
    {
        SCK = 0;
        SDA = temp&0x01;
        temp>>=1;
        SCK=1;
    }
}  
  
//
void Write_Ds1302_Byte( unsigned char address,unsigned char dat )    
{
    RST=0;  _nop_();
    SCK=0;  _nop_();
    RST=1;  _nop_();  
    Write_Ds1302(address);  
    Write_Ds1302(dat);      
    RST=0;
}
  
//
unsigned char Read_Ds1302_Byte ( unsigned char address )
{
    unsigned char i,temp=0x00;
    RST=0;  _nop_();
    SCK=0;  _nop_();
    RST=1;  _nop_();
    Write_Ds1302(address);
    for (i=0;i<8;i++)  
    {      
        SCK=0;
        temp>>=1;  
        if(SDA)
        temp|=0x80;
        SCK=1;
    }
    RST=0;  _nop_();
    SCK=0;  _nop_();
    SCK=1;  _nop_();
    SDA=0;  _nop_();
    SDA=1;  _nop_();
    return (temp);          
}
~~~

## smg_ca.c

~~~C
#include "absacc.h"
#include "STC15F2K60S2.h"
#include "intrins.h"
  
void Delay1ms()     //@11.0592MHz
{
    unsigned char i, j;
  
    _nop_();
    _nop_();
    i = 11;
    j = 190;
    do
    {
        while (--j);
    } while (--i);
}
  
void DisplaySMG(unsigned char pos, unsigned char dat){
    XBYTE[0xE000] = 0xff;
    XBYTE[0xC000] = 0x01 << pos;
    XBYTE[0xE000] = dat;
}
  
void Display_All(unsigned char dat){
    XBYTE[0xE000] = 0xff;
    XBYTE[0xC000] = dat;
}
~~~

### # include "absacc.h"的解释：

是用于绝对地址访问的标准头文件。它提供的一组宏定义，允许直接访问特定的内存地址。

#### 硬件寄存器操作：

通过预定义的宏直接访问：

~~~C
XBYTE[0x8000] = 0xFF;  // 向外部地址0x8000写入数据
unsigned char val = XBYTE[0x8000];  // 从外部地址读取数据
~~~

#### 内存类型区分：

1. 支持不同存储空间的访问：

- `CBYTE`（CODE区）
- `DBYTE`（DATA区）
- `XBYTE`（XDATA区）

#### 兼容性说明：

- 主要用于8051架构开发
- 现代嵌入式开发**更推荐使用指针**（如 `volatile unsigned char xdata *`）
- 不同编译器实现可能略有差异
