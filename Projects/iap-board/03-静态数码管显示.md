# 代码展示

~~~C
#include <reg52.h>
#include <intrins.h>
  
void Delay1000ms(void)  //@11.0592MHz
{
    unsigned char data i, j, k;
  
    _nop_();
    _nop_();
    i = 43;
    j = 6;
    k = 203;
    do
    {
        do
        {
            while (--k);
        } while (--j);
    } while (--i);
}

unsigned char code SMG_duanma[18] = {
    0xc0,0xf9,0xa4,0xb0,0x99,0x92,
    0x82,0xf8,0x80,0x90,0x88,0x80,
    0xc6,0xc0,0x86,0x8e,0xbf,0x7f
};//进行段选的代码

void InitHC138(unsigned char n){//通过74HC138选择数码管模式
    switch(n){
        case 4:
            P2 = (P2 & 0x1f) | 0x80;
            break;
        case 5:
            P2 = (P2 & 0x1f) | 0xa0;
            break;
        case 6:
            P2 = (P2 & 0x1f) | 0xc0;
            break;
        case 7:
            P2 = (P2 & 0x1f) | 0xe0;
            break;
    }
}

  

void ShowSMG_Bit(unsigned char dat,unsigned char pos){
    InitHC138(6);//数码管位选模式
    P0 = 0x01 << pos;
    InitHC138(7);//数码管段选模式
    P0 = dat;
}

void SMG_Static(){
    unsigned char i,j;
    for(i = 0; i<8; i++){
        for(j=0;j<10;j++){
            ShowSMG_Bit(SMG_duanma[j],i);
            Delay1000ms();
        }
    }
    for(j = 0 ; j<8; j++){
        InitHC138(6);
        P0 = 0xff;
        InitHC138(7);
        P0 = SMG_duanma[j];
    }
}//实际实现的功能其实在这里

void main(){
    while(1){
        SMG_Static();
        Delay1000ms();
    }
}
~~~
